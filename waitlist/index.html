async function submitWaitlist() {

  const email = document.getElementById('wl-email')?.value?.trim();

  const errEl = document.getElementById('wl-err');

  const btn   = document.getElementById('wl-submit-btn');


  if (!email || !email.includes('@')) {

    if (errEl) { errEl.textContent = 'Zadaj platný email.'; errEl.style.display = 'block'; }

    return;

  }

  if (errEl) errEl.style.display = 'none';

  if (btn) { btn.disabled = true; btn.textContent = 'Ukladám...'; }


  try {

    const { data: existing } = await _sb

      .from('waitlist').select('id').eq('email', email).maybeSingle();


    if (existing) {

      document.getElementById('wl-form').style.display = 'none';

      document.getElementById('wl-success').style.display = 'block';

      return;

    }


    const { error } = await _sb.from('waitlist').insert({

      email,

      role: _wlRole,

      source: 'landing',

      created_at: new Date().toISOString(),

    });

    if (error) throw error;


    document.getElementById('wl-form').style.display = 'none';

    document.getElementById('wl-success').style.display = 'block';


  } catch (err) {

    if (errEl) { errEl.textContent = 'Niečo sa pokazilo. Skús znova.'; errEl.style.display = 'block'; }

    if (btn) { btn.disabled = false; btn.textContent = 'Zaregistrovať sa'; }

  }

}
